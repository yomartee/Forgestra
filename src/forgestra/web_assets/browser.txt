// Local-only browser acceptance against an authored GraphQL protocol fixture.
import { chromium } from "playwright-core";
import { spawn } from "node:child_process";
import { createServer } from "node:net";
import { cp, writeFile } from "node:fs/promises";
import assert from "node:assert/strict";
import { resources, operations } from "../src/generated/domain.ts";
import { startBackend, USER_A, USER_B } from "./backend-fixture.mjs";
const backend = await startBackend();
const socket=createServer();await new Promise(resolve=>socket.listen(0,"127.0.0.1",resolve));const port=socket.address().port;await new Promise(resolve=>socket.close(resolve));
const base=`http://127.0.0.1:${port}`;
await cp(".next/static",".next/standalone/.next/static",{recursive:true});
let output="";
const server=spawn(process.execPath,[".next/standalone/server.js"],{env:{...process.env,PORT:String(port),HOSTNAME:"127.0.0.1",HASURA_GRAPHQL_ENDPOINT:backend.endpoint,NEXT_TELEMETRY_DISABLED:"1"},windowsHide:true,stdio:["ignore","pipe","pipe"]});
server.stdout.on("data",b=>{output+=b;});server.stderr.on("data",b=>{output+=b;});
server.on("error",error=>{output+="server process error: "+error.name;});
let browser;let page;let backendClosed=false;const diagnostics=[];
const checks=[];
try {
  let ready=false;
  const deadline=Date.now()+60000;
  while(Date.now()<deadline){try{if((await fetch(base,{signal:AbortSignal.timeout(1000)})).ok){ready=true;break;}}catch{}await new Promise(resolve=>setTimeout(resolve,100));}
  assert.ok(ready,"Production server did not start: "+output);
  const op=operations.find(o=>o.kind==="list");
  const probe=await fetch(base+"/api/graphql",{method:"POST",headers:{origin:base,"content-type":"application/json",cookie:`forgestra_user=${backend.token(USER_A)}`},body:JSON.stringify({operationName:op.name,query:op.query,variables:{limit:25,offset:0}})});
  diagnostics.push({probeStatus:probe.status,body:await probe.text()});
  browser=await chromium.launch({executablePath:process.env.FORGESTRA_BROWSER,headless:true,timeout:30000,args:["--disable-background-networking","--disable-sync","--disable-component-update","--host-resolver-rules=MAP * ~NOTFOUND, EXCLUDE 127.0.0.1, EXCLUDE localhost"]});
  const context=await browser.newContext();
  page=await context.newPage();
  page.on("pageerror",error=>diagnostics.push({pageError:error.message}));
  page.on("response",async response=>{if(new URL(response.url()).pathname==="/api/graphql")diagnostics.push({status:response.status(),origin:response.request().headers()["origin"],body:await response.text().catch(()=>null)});});
  const external=[];
  await page.route("**/*",route=>{const host=new URL(route.request().url()).hostname;if(host!=="127.0.0.1"){external.push(host);return route.abort();}return route.continue();});
  const resource=resources[0];
  const session=async user=>{await context.clearCookies();await context.addCookies([{name:"forgestra_user",value:backend.token(user),url:base,httpOnly:true,sameSite:"Strict"}]);};
  await session(USER_A);
  await page.goto(base);await page.getByRole("link",{name:resource.label+" →"}).click();
  await page.getByText("No records yet.",{exact:true}).waitFor();checks.push("root-to-list");
  await page.getByRole("link",{name:"Create record"}).click();
  for(const f of resource.fields){const field=page.locator(`[name="${f.name}"]`);if(f.type==="boolean"){if(f.nullable)await field.selectOption("true");else await field.check();}else await field.fill(f.type==="number"?"1.5":"Browser record");}
  await page.getByRole("button",{name:"Save record"}).click();await page.getByText("Saved successfully.",{exact:true}).waitFor();checks.push("create-through-ui");
  await page.getByRole("link",{name:"← Back to list"}).click();await page.getByRole("link",{name:"Edit",exact:true}).waitFor();
  await page.getByRole("link",{name:"Edit",exact:true}).click();
  await page.waitForURL(`${base}/${resource.name}/*/edit`);
  await page.getByRole("heading",{name:"Edit record"}).waitFor();const editURL=page.url();
  const first=resource.fields[0];const field=page.locator(`[name="${first.name}"]`);
  if(first.type==="string")await field.fill("Edited browser record");else if(first.type==="number")await field.fill("2.5");else if(first.nullable)await field.selectOption("false");else await field.uncheck();
  await page.getByRole("button",{name:"Save record"}).click();await page.getByText("Saved successfully.",{exact:true}).waitFor();checks.push("dynamic-edit-through-ui");
  assert.equal(await page.evaluate(()=>document.cookie),"");checks.push("http-only-session");
  await session(USER_B);await page.goto(editURL);await page.getByRole("alert").filter({hasText:"Record unavailable or access denied"}).waitFor();checks.push("cross-user-edit-denied");
  await context.clearCookies();await page.goto(base+"/"+resource.name);await page.getByRole("alert").filter({hasText:"Sign in required"}).waitFor();checks.push("missing-session-denied");
  await session(USER_A);await page.goto(base+"/"+resource.name);await page.getByRole("link",{name:"Edit",exact:true}).waitFor();
  await page.screenshot({path:"browser.png",fullPage:true});
  page.on("dialog",dialog=>dialog.accept());await page.getByRole("button",{name:"Delete",exact:true}).click();await page.getByText("No records yet.",{exact:true}).waitFor();checks.push("delete-through-ui");
  await backend.close();backendClosed=true;await page.reload();await page.getByRole("alert").filter({hasText:"Request denied or backend unavailable"}).waitFor();checks.push("backend-failure-state");
  assert.deepEqual(external,[]);checks.push("no-external-browser-requests");
  assert.equal(backend.adminHeaders(),0);
  const result={status:"PASSED_LOCAL",mode:"AUTHORED_GRAPHQL_PROTOCOL_FIXTURE",checks,real_hasura:"NOT_RUN",container:"NOT_RUN",external_requests:external,admin_headers:backend.adminHeaders()};
  await writeFile("browser-result.json",JSON.stringify(result,null,2)+"\n");
  console.log(JSON.stringify(result));
} catch(error) {
  await writeFile("browser-failure.json",JSON.stringify({message:error.message,body:page?await page.locator("body").innerText().catch(()=>""):"",diagnostics},null,2));
  throw error;
} finally {
  if(browser)await browser.close();
  server.kill();await new Promise(resolve=>{if(server.exitCode!==null)resolve();else server.once("exit",resolve);});
  if(!backendClosed)await backend.close();
  await writeFile("browser-server.log",output);
}
